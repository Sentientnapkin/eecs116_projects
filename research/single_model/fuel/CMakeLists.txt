cmake_minimum_required(VERSION 3.20)
project(gp_msckf_fuel C)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(VIOBENCH_NATIVE_DIR "" CACHE PATH "")
include(${VIOBENCH_NATIVE_DIR}/viobench_reactor.cmake)
viobench_solution(solution solution.c)
target_link_libraries(solution PRIVATE m)
# optional per-variant compile flags, selected through -D markers placed in CMAKE_C_FLAGS by run_variant --defines
if(CMAKE_C_FLAGS MATCHES "FUEL_O3")
    target_compile_options(solution PRIVATE -O3)
endif()
if(CMAKE_C_FLAGS MATCHES "FUEL_OS")
    target_compile_options(solution PRIVATE -Os)
endif()
if(CMAKE_C_FLAGS MATCHES "FUEL_OZ")
    target_compile_options(solution PRIVATE -Oz)
endif()
if(CMAKE_C_FLAGS MATCHES "FUEL_FASTMATH")
    target_compile_options(solution PRIVATE -ffast-math)
endif()
if(CMAKE_C_FLAGS MATCHES "FUEL_NOSIMD")
    target_compile_options(solution PRIVATE -mno-simd128)
endif()
if(CMAKE_C_FLAGS MATCHES "FUEL_LTO")
    target_compile_options(solution PRIVATE -flto)
    target_link_options(solution PRIVATE -flto)
endif()
if(CMAKE_C_FLAGS MATCHES "FUEL_UNROLL")
    target_compile_options(solution PRIVATE -funroll-loops)
endif()
